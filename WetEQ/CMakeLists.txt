cmake_minimum_required(VERSION 3.14.0)

option(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES "Enable VST 3 Plug-in Examples" OFF)
option(SMTG_ENABLE_VST3_HOSTING_EXAMPLES "Enable VST 3 Hosting Examples" OFF)

set(CMAKE_OSX_DEPLOYMENT_TARGET 10.13 CACHE STRING "")

set(vst3sdk_SOURCE_DIR "../vst3sdk")
if(NOT vst3sdk_SOURCE_DIR)
    message(FATAL_ERROR "Path to VST3 SDK is empty!")
endif()

project(WetEQ
    # This is your plug-in version number. Change it here only.
    VERSION 1.1.1.0
    DESCRIPTION "WetEQ VST 3 Plug-in"
)

set(SMTG_VSTGUI_ROOT "${vst3sdk_SOURCE_DIR}")

add_subdirectory(${vst3sdk_SOURCE_DIR} ${PROJECT_BINARY_DIR}/vst3sdk)
smtg_enable_vst3_sdk()

smtg_add_vst3plugin(WetEQ
    source/version.h
    source/weteqcids.h
    source/wetcore.h
    source/eqengine.h
    source/eqengine.cpp
    source/weteqprocessor.h
    source/weteqprocessor.cpp
    source/weteqcontroller.h
    source/weteqcontroller.cpp
    source/weteqentry.cpp
    source/ledmeterview.h
    source/ledmeterview.cpp
    source/steppedknob.h
    source/steppedknob.cpp
    source/customviewcreator.h
    source/customviewcreator.cpp
)

#- VSTGUI Wanted ----
if(SMTG_ENABLE_VSTGUI_SUPPORT)
    target_sources(WetEQ
        PRIVATE
            resource/weteqeditor.uidesc
    )
    target_link_libraries(WetEQ
        PRIVATE
            vstgui_support
    )
    # Everything the editor loads at runtime. The knob filmstrips and the
    # backplate are generated by tools/make-assets.py from the panel art.
    smtg_target_add_plugin_resources(WetEQ
        RESOURCES
            "resource/weteqeditor.uidesc"
            "resource/backplate.png"
            "resource/knob_gain.png"
            "resource/knob_hpf.png"
            "resource/knob_lpf.png"
            "resource/knob_hf_gain.png"
            "resource/knob_hf_freq.png"
            "resource/knob_hmf_gain.png"
            "resource/knob_hmf_freq.png"
            "resource/knob_lmf_gain.png"
            "resource/knob_lmf_freq.png"
            "resource/knob_lf_gain.png"
            "resource/knob_lf_freq.png"
    )
endif(SMTG_ENABLE_VSTGUI_SUPPORT)
# -------------------

target_link_libraries(WetEQ
    PRIVATE
        sdk
)

smtg_target_configure_version_file(WetEQ)

if(SMTG_MAC)
    smtg_target_set_bundle(WetEQ
        BUNDLE_IDENTIFIER org.yonie.vst3.weteq
        COMPANY_NAME "Yonie"
    )
    smtg_target_set_debug_executable(WetEQ
        "/Applications/VST3PluginTestHost.app"
        "--pluginfolder;$(BUILT_PRODUCTS_DIR)"
    )
elseif(SMTG_WIN)
    target_sources(WetEQ PRIVATE
        resource/win32resource.rc
    )
    if(MSVC)
        set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT WetEQ)

        smtg_target_set_debug_executable(WetEQ
            "$(ProgramW6432)/Steinberg/VST3PluginTestHost/VST3PluginTestHost.exe"
            "--pluginfolder \"$(OutDir)/\""
        )
    endif()
elseif(SMTG_LINUX)
    # Linux uses default settings
endif(SMTG_MAC)
